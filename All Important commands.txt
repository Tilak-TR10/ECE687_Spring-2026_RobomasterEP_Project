1. For Simulation Run the Following docker Run:
sudo docker run -it --rm \
  --network=host \
  --pid=host \
  --ipc=host \
  --name="dji_robomaster_ros" \
  -v $HOME/project:/project:rw \
  -v $HOME/ros_startup.sh:/ros_startup.sh \
  dji_robomaster_ros:1.0

2. Robomaster Docker Run:
sudo docker run -it --rm\
  --network=host \
  --pid=host \
  --ipc=host \
  --name="dji_robomaster_ros" \
  -v $HOME/project:/project:rw \
  -v $HOME/ros_startup.sh:/ros_startup.sh \
  --env ROS_DISCOVERY_SERVER=192.168.0.2:11811 \
  --env ROS_SUPER_CLIENT=TRUE \

/robot10/target_arm_position \
geometry_msgs/msg/Point \
"{x: 0.10, y: 0.0, z: 0.05}"

publishing #1: geometry_msgs.msg.Point(x=0.1, y=0.0, z=0.05)

2. Extend Arm Farward
ros2 topic pub \
/robot10/target_arm_position \
geometry_msgs/msg/Point \
"{x: 0.20, y: 0.0, z: 0.05}"

publishing #1: geometry_msgs.msg.Point(x=0.2, y=0.0, z=0.05)

3. Move Arm Up (To Rmeove Stick)
ros2 topic pub \
/robot10/target_arm_position \
geometry_msgs/msg/Point \
"{x: 0.20, y: 0.0, z: 0.15}"

publishing #1: geometry_msgs.msg.Point(x=0.2, y=0.0, z=0.15)

4. Move Arm Back to initial Pose
ros2 topic pub \
/robot10/target_arm_position \
geometry_msgs/msg/Point \
"{x: 0.10, y: 0.0, z: 0.05}"
